"""IKA-439: train capacity candidates from nothing (CLAUDE.md "next generation" command + one flag).

    trainchain.py <tag> <extra flags...>   -> models/value-<tag>.pt (seed 10), value-<tag>-s1.pt (seed 11)
Each: wait for free mem >= 9 GB and GPU < 7 GB, take train.lock (mkdir, pid inside), run through heavy.
Env: EST (minutes for heavy), SEEDS ("0,1").
"""
import os
import shutil
import subprocess
import sys
import threading
import time
from pathlib import Path

import psutil

H = Path("C:/tmp/ika439")
M = Path("C:/Users/Ikazuchi/repos/pokeuraou")
WT = M / ".claude/worktrees/agent-abab28403d0bd0449"
PY = str(M / ".venv/Scripts/python.exe")
HEAVY = "C:/tmp/pokeuraou-machine/heavy.py"
LOCK = Path("C:/tmp/pokeuraou-machine/train.lock")
DATA = M / "data/selfplay-mc0123456-encoded.npz"


def used():
    o = subprocess.run(["nvidia-smi", "--query-gpu=memory.used", "--format=csv,noheader,nounits"],
                       capture_output=True, text=True, check=False).stdout.strip()
    return int(o or 0)


def say(t):
    with open(H / "trainchain.log", "a", encoding="utf-8", newline="\n") as fh:
        fh.write(f"{time.strftime('%H:%M:%S')} {t}\n")


def run_one(tag, k, extra, est):
    out = H / "models" / (f"value-{tag}.pt" if k == 0 else f"value-{tag}-s{k}.pt")
    if out.exists():
        say(f"{out.name} exists, skipped")
        return
    while True:
        if psutil.virtual_memory().available >= 9 * 2**30 and used() < 7 * 1024:
            try:
                LOCK.mkdir()
            except FileExistsError:
                pass
            else:
                (LOCK / "pid").write_bytes(f"{os.getpid()}\n".encode())
                break
        time.sleep(60)
    say(f"start {out.name}")
    env = dict(os.environ, PYTHONPATH=str(WT / "src"), PYTHONUNBUFFERED="1")
    cmd = [PY, str(WT / "tools/train_value.py"), "--data", str(DATA), "--state-inputs", "--bind-inputs",
           "--aux-weight", "5", "--aux-targets", "search:1,ahead2:5", "--epochs", "30", "--lr", "2e-3",
           "--keep", "best", "--split-seed", "0", "--holdout", "0.15", "--seed", str(10 + k),
           "--out", str(out)] + extra
    before = used()
    peak = [before]
    stop = threading.Event()

    def sample():
        while not stop.is_set():
            peak[0] = max(peak[0], used())
            stop.wait(2)

    th = threading.Thread(target=sample, daemon=True)
    th.start()
    t = time.time()
    try:
        with (H / "models" / f"{out.stem}.log").open("w", encoding="utf-8", newline="\n") as fh:
            fh.write(" ".join(cmd) + "\n")
            fh.flush()
            rc = subprocess.run([PY, HEAVY, "--agent", "ika-439", "--gpu", "--cores", "3", "--est-min", str(est),
                                 "--wait-max", "720", "--why", f"IKA-439 train {out.stem}", "--"] + cmd,
                                cwd=str(WT), env=env, stdout=fh, stderr=subprocess.STDOUT).returncode
    finally:
        stop.set()
        th.join()
        shutil.rmtree(LOCK, ignore_errors=True)
    say(f"{out.name} rc={rc} {time.time() - t:.0f}s GPU before {before} MB peak {peak[0]} MB")
    if rc:
        sys.exit(rc)


if __name__ == "__main__":
    tag = sys.argv[1]
    est = int(os.environ.get("EST", "100"))
    seeds = [int(s) for s in os.environ.get("SEEDS", "0,1").split(",")]
    for k in seeds:
        run_one(tag, k, sys.argv[2:], est)
    say(f"chain {tag} end")
