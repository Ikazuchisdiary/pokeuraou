"""IKA-439 stage 2 A: presolve value-mc6w15x2 selections, null (40 pairs), SPRT(0,10) vs value-mc6 x2.
Each through heavy (GPU); GPU used < 7 GB first. Log: C:/tmp/ika439/chainA.log
"""
import os
import subprocess
import sys
import time
from pathlib import Path

H = Path("C:/tmp/ika439")
M = Path("C:/Users/Ikazuchi/repos/pokeuraou")
WT = M / ".claude/worktrees/agent-abab28403d0bd0449"
PY = str(M / ".venv/Scripts/python.exe")
HEAVY = "C:/tmp/pokeuraou-machine/heavy.py"
env = dict(os.environ, IKA439_ROOT=str(WT), PYTHONPATH=str(WT / "src"),
           POKEURAOU_RUST_NODE_BIN=str(WT / "rust/target/release/pokeuraou-damage.exe"))


def used() -> int:
    out = subprocess.run(["nvidia-smi", "--query-gpu=memory.used", "--format=csv,noheader,nounits"],
                         capture_output=True, text=True, check=False).stdout.strip()
    return int(out or 0)


def say(t):
    with open(H / "chainA.log", "a", encoding="utf-8", newline="\n") as fh:
        fh.write(f"{time.strftime('%H:%M:%S')} {t}\n")


def heavy(why, est, cores, cmin, cmd, log):
    while used() >= 7 * 1024:
        time.sleep(30)
    with open(log, "w", encoding="utf-8", newline="\n") as fh:
        return subprocess.run([PY, HEAVY, "--agent", "ika-439", "--gpu", "--cores", str(cores), "--cores-min",
                               str(cmin), "--est-min", str(est), "--why", why, "--", *cmd],
                              env=env, stdout=fh, stderr=subprocess.STDOUT).returncode


store = H / "store-mc6w15x2"
if len(list(store.glob("*.json"))) < 2145:
    rc = heavy("IKA-439 presolve value-mc6w15x2 selections", 10, 8, 8,
               [PY, str(H / "presolve.py"), "value-mc6w15x2", str(store), "{cores}",
                str(H / "w15/value-mc6w15.pt"), str(H / "w15/value-mc6w15-s1.pt")],
               H / "presolve-w15.out")
    say(f"presolve rc={rc} files={len(list(store.glob('*.json')))}")
    if rc:
        sys.exit(rc)
for m in (sys.argv[1:] or ["null", "w15-vs-mc6"]):
    if (H / "matches" / m).exists():
        say(f"{m} exists")
        continue
    rc = heavy(f"IKA-439 board {m}", 25, 12, 8, [PY, str(H / "board.py"), m, "{cores}", "1"], H / f"chain-{m}.out")
    say(f"{m} rc={rc}")
    if rc:
        sys.exit(rc)
say("chainA end")
